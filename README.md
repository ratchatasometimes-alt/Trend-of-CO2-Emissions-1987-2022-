## ตัวอย่างการใช้งาน

```python
import pandas as pd

df = pd.read_csv("thailand_co2_emission_1987_2022.csv")

# ยอดรวมรายปี
yearly = df.groupby("year")["emissions_tons"].sum()

# สัดส่วนตามภาคส่วน
by_source = df.groupby("source")["emissions_tons"].sum()
print((by_source / by_source.sum()).round(3))

# แนวโน้มรายปีแยกตามเชื้อเพลิง
trend = df.pivot_table(index="year", columns="fuel_type",
                       values="emissions_tons", aggfunc="sum")
trend.plot()
```
